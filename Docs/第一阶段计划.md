# 核心控制与流转层 (Core Control) 设计文档

## 1. 架构选型与依赖库

- **包管理与虚拟环境:** `uv`
- **事件调度中心 (Pub-Sub):** `blinker`
- **数据验证与上下文 (Context):** `pydantic`
- **日志系统:** Python内置 `logging.config.dictConfig`
- **接口约束与切面工具:** Python内置 `abc` 与 `functools`（用于构建装饰器）

### 3.2 接口与自定义异常 (`interfaces.py` & `exceptions.py`)

使用 `@abstractmethod` 装饰器强制实现依赖倒置原则 (DIP)。

```python
# core_control/exceptions.py
class ZSimBaseException(Exception): pass
class ActionExecutionError(ZSimBaseException): pass

# core_control/interfaces.py
from abc import ABC, abstractmethod
from typing import Any

class IResourceValidator(ABC):
    @abstractmethod
    def can_execute(self, action_id: str, state: Any) -> bool:
        """子类必须实现此方法，否则实例化时抛出 TypeError"""
        pass
```

### 3.3 全局事件与切面装饰器 (`dispatcher.py` & `decorators.py`)

将异常捕获与事件广播的职责从主业务代码中剥离（单一职责原则）。

```python
# core_control/dispatcher.py
from blinker import Namespace

_zsim_signals = Namespace()
on_tick = _zsim_signals.signal('on_tick')
on_action_request = _zsim_signals.signal('on_action_request')
on_action_start = _zsim_signals.signal('on_action_start')
on_error_raised = _zsim_signals.signal('on_error_raised')
on_combat_end = _zsim_signals.signal('on_combat_end')

# core_control/decorators.py
import functools
import logging
from typing import Callable, Any
from .dispatcher import on_error_raised
from .exceptions import ZSimBaseException

logger = logging.getLogger("zsim.Core.Decorators")

def emit_on_error(func: Callable) -> Callable:
    """装饰器：自动捕获执行异常，通过 blinker 广播，并安全向上抛出"""
    @functools.wraps(func)
    def wrapper(self, *args, **kwargs) -> Any:
        try:
            return func(self, *args, **kwargs)
        except Exception as e:
            logger.error(f"Error captured in {func.__name__}: {str(e)}")
            on_error_raised.send(self, error=e)
            raise e
    return wrapper
```

### 3.4 带有校验装饰器的上下文快照 (`game_state.py`)

使用 Pydantic 的 `@field_validator` 确保写入数据的逻辑绝对合法。

```python
# core_control/game_state.py
from pydantic import BaseModel, Field, field_validator

class GameState(BaseModel):
    current_tick: int = Field(default=0, description="当前模拟tick")
    max_ticks: int = Field(default=18000, description="最大限制tick")
    is_running: bool = Field(default=True, description="模拟器运行状态")

    @field_validator('current_tick', 'max_ticks')
    @classmethod
    def check_non_negative(cls, value: int) -> int:
        """装饰器：拦截负数tick赋值"""
        if value < 0:
            raise ValueError("Tick count cannot be negative.")
        return value
```

### 3.5 Tick驱动器 (`event_bus.py`)

```python
# core_control/event_bus.py
import logging
from .dispatcher import on_tick, on_combat_end
from .game_state import GameState

logger = logging.getLogger("zsim.Core.EventBus")

class EventBus:
    def __init__(self, state: GameState):
        self.state = state

    def run_simulation(self):
        logger.info(f"Simulation started. Max ticks: {self.state.max_ticks}")
        
        while self.state.is_running and self.state.current_tick < self.state.max_ticks:
            current = self.state.current_tick
            logger.debug(f"[TICK] Processing tick: {current}")
            
            # 广播 Tick 事件，下游组件使用 @on_tick.connect 监听此事件
            on_tick.send(self, tick=current)
            self.state.current_tick += 1

        reason = "Max ticks reached" if self.state.current_tick >= self.state.max_ticks else "Simulation stopped"
        logger.info(f"Simulation ended at tick {self.state.current_tick}. Reason: {reason}")
        on_combat_end.send(self, reason=reason)
```

### 3.6 APL 解析与管理器 (`apl_manager.py`)

应用 `@emit_on_error` 装饰器，使业务代码无需编写 `try...except` 即可实现标准错误上报。

```python
# core_control/apl_manager.py
import json
import logging
from typing import List, Dict, Optional
from .interfaces import IResourceValidator
from .dispatcher import on_action_request, on_action_start
from .decorators import emit_on_error
from .exceptions import ActionExecutionError

logger = logging.getLogger("zsim.Core.APLManager")

class APLManager:
    def __init__(self, validator: IResourceValidator):
        self.validator = validator
        self.action_queue: List[Dict] = []
        self.last_successful_action_id: Optional[str] = None

    def load_from_json(self, file_path: str):
        logger.info(f"Loading APL from: {file_path}")
        with open(file_path, 'r', encoding='utf-8') as f:
            data = json.load(f)
            self.action_queue = data.get("scenarioList", [])[0].get("data", {}).get("tracks", [])[0].get("actions", [])
        logger.debug(f"Loaded {len(self.action_queue)} actions.")

    @emit_on_error
    def process_next_action(self, state: 'GameState'):
        """装饰器负责接管任何此处抛出的异常并广播"""
        if not self.action_queue:
            return False

        next_action = self.action_queue.pop(0)
        action_id = next_action.get("id")
        
        logger.debug(f"Requesting action execution: {action_id}")
        on_action_request.send(self, action_id=action_id)

        if self.validator.can_execute(action_id, state):
            self.last_successful_action_id = action_id
            logger.info(f"Action validated and started: {action_id}")
            on_action_start.send(self, action_id=action_id)
            return True
        else:
            error_msg = f"当前指令 {action_id} 无法释放，前一个指令是 {self.last_successful_action_id}"
            # 直接抛出自定义异常，由 @emit_on_error 装饰器进行处理和广播
            raise ActionExecutionError(error_msg)
```
